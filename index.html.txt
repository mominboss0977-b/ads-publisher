  <script>
    const langData = {
      en: {
        title: "CybernetX Ads",
        tagline: "Free Ad Publishing for All",
        formTitle: "Post an Ad",
        galleryTitle: "Published Ads"
      },
      hi: {
        title: "साइबरनेटएक्स विज्ञापन",
        tagline: "सभी के लिए मुफ्त विज्ञापन प्रकाशन",
        formTitle: "विज्ञापन पोस्ट करें",
        galleryTitle: "प्रकाशित विज्ञापन"
      }
    };

    function setLang(lang) {
      document.getElementById("title").innerText = langData[lang].title;
      document.getElementById("tagline").innerText = langData[lang].tagline;
      document.getElementById("formTitle").innerText = langData[lang].formTitle;
      document.getElementById("galleryTitle").innerText = langData[lang].galleryTitle;
    }

    document.getElementById("adForm").addEventListener("submit", function(e) {
      e.preventDefault();
      const title = document.getElementById("adTitle").value;
      const desc = document.getElementById("adDesc").value;
      const cat = document.getElementById("adCat").value;
      const link = document.getElementById("adLink").value;
      const image = document.getElementById("adImage").files[0];

      const reader = new FileReader();
      reader.onload = function() {
        const adHTML = `
          <div class="ad-card">
            <h3>${title}</h3>
            <p><strong>Category:</strong> ${cat}</p>
            <p>${desc}</p>
            ${link ? `<p><a href="${link}" target="_blank">Visit Link</a></p>` : ""}
            ${image ? `<img src="${reader.result}" alt="Ad Image"/>` : ""}
          </div>
        `;
        document.getElementById("adsGallery").innerHTML += adHTML;
        document.getElementById("adForm").reset();
      };
      if (image) reader.readAsDataURL(image);
      else reader.onload();
    });
  </script>
